cmake_minimum_required(VERSION 3.16)
project(ChihiroPackLauncher LANGUAGES C)
add_executable(ChihiroPack launcher.c)
set_property(TARGET ChihiroPack PROPERTY C_STANDARD 11)
if(MINGW)
 target_compile_options(ChihiroPack PRIVATE -O2 -Wall -Wextra)
 target_link_options(ChihiroPack PRIVATE -municode -static -s)
 target_compile_options(ChihiroPack PRIVATE -municode)
endif()
if(MSVC)
 set_property(TARGET ChihiroPack PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
 target_compile_options(ChihiroPack PRIVATE /O2 /W4)
endif()
